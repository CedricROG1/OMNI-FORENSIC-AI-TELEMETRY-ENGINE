Sub FormatSnapshotVertical()
    ' Applies exact Width (22.5 cm) and Length (19.5 cm)
    ApplyImageFormatting CentimetersToPoints(17.8), CentimetersToPoints(22.8)
End Sub

Sub FormatSnapshotHorizontal()
    ' Applies exact Width (22.7 cm) and Length (12.5 cm)
    ApplyImageFormatting CentimetersToPoints(26.5), CentimetersToPoints(14.7)
End Sub

Private Sub ApplyImageFormatting(TargetWidth As Single, TargetHeight As Single)
    If Selection.InlineShapes.Count = 0 Then
        MsgBox "Please click on a snapshot first.", vbExclamation
        Exit Sub
    End If

    Dim img As InlineShape
    Dim b As Border
    Set img = Selection.InlineShapes(1)

    ' 1. Unlock aspect ratio to force exact Width AND Length
    img.LockAspectRatio = msoFalse
    img.Width = TargetWidth
    img.Height = TargetHeight

    ' 2. Apply professional black border
    For Each b In img.Borders
        b.LineStyle = wdLineStyleSingle
        b.LineWidth = wdLineWidth075pt
        b.Color = wdColorBlack
    Next b

    ' 3. Apply exact 0pt spacing, remove indents, and align to the 0 margin mark
    With Selection.ParagraphFormat
        .SpaceBefore = 0
        .SpaceAfter = 0
        .LineSpacingRule = wdLineSpaceSingle
        .Alignment = wdAlignParagraphLeft
        .LeftIndent = 0
        .FirstLineIndent = 0
    End With
End Sub
